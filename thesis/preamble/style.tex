%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
%  Visual style — reproduces the department's thesis look from the reference
%  PDF: "N Chapter" above the chapter title, and a running header reading
%  "CHAPTER N. TITLE".
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% --- Palette -------------------------------------------------------------
\definecolor{uamobblue}{RGB}{20,60,110}
\definecolor{rulegray}{RGB}{120,120,120}
\definecolor{codebg}{RGB}{247,247,249}
\definecolor{codekw}{RGB}{20,60,140}
\definecolor{codestr}{RGB}{140,40,40}
\definecolor{codecmt}{RGB}{100,110,100}
\definecolor{todoborder}{RGB}{176,58,46}
\definecolor{todobg}{RGB}{253,245,244}

% --- Chapter headings ----------------------------------------------------
\titleformat{\chapter}[display]
  {\normalfont\filright}
  {\hfill\Huge\bfseries\color{uamobblue}\thechapter\;\normalfont\Large Chapter}
  {6pt}
  {\hfill\Huge\bfseries}
  [\vspace{4pt}{\color{rulegray}\titlerule[0.8pt]}]
\titlespacing*{\chapter}{0pt}{10pt}{34pt}

\titleformat{\section}
  {\normalfont\large\bfseries\color{uamobblue}}{\thesection}{0.8em}{}
\titleformat{\subsection}
  {\normalfont\normalsize\bfseries}{\thesubsection}{0.8em}{}

% --- Running headers -----------------------------------------------------
%  fancyhdr warns if the header box is shorter than its content; 15pt clears
%  the 12pt body font without leaving a gap.
\setlength{\headheight}{15pt}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\nouppercase{\leftmark}}
\fancyhead[R]{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\small\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}

% --- Table of contents depth --------------------------------------------
\setcounter{tocdepth}{2}
\setcounter{secnumdepth}{3}

% --- Listings ------------------------------------------------------------
\lstdefinestyle{thesis}{
  backgroundcolor=\color{codebg},
  basicstyle=\ttfamily\scriptsize,
  keywordstyle=\color{codekw}\bfseries,
  stringstyle=\color{codestr},
  commentstyle=\color{codecmt}\itshape,
  numbers=left,
  numberstyle=\ttfamily\tiny\color{rulegray},
  numbersep=8pt,
  frame=single,
  rulecolor=\color{rulegray},
  framesep=5pt,
  breaklines=true,
  breakatwhitespace=false,
  showstringspaces=false,
  tabsize=2,
  captionpos=b,
  columns=fullflexible,
  keepspaces=true,
  xleftmargin=14pt,
  aboveskip=12pt,
  belowskip=12pt,
}
\lstset{style=thesis}
\renewcommand{\lstlistingname}{Listing}
