%  Visual style: a centred "Chapter N" label above the chapter title, and a
%  running header reading "Chapter N. Title".
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

% --- Palette -------------------------------------------------------------
\definecolor{uamobblue}{RGB}{20,60,110}
\definecolor{rulegray}{RGB}{120,120,120}
\definecolor{codebg}{RGB}{247,247,249}
\definecolor{codekw}{RGB}{20,60,140}
\definecolor{codestr}{RGB}{140,40,40}
\definecolor{codecmt}{RGB}{100,110,100}
\definecolor{todoborder}{RGB}{176,58,46}
\definecolor{todobg}{RGB}{253,245,244}

% --- Chapter headings ----------------------------------------------------
%  Numbered: "Chapter N" on its own line, the title under it, a rule below.
%  Unnumbered (Abstract, Contents, Conclusion, Bibliography...): the same
%  title and rule without the label, so every chapter opening lines up.
\titleformat{\chapter}[display]
  {\normalfont\centering}
  {\Large\scshape\color{uamobblue}\chaptertitlename~\thechapter}
  {8pt}
  {\huge\bfseries}
  [\vspace{6pt}{\color{rulegray}\titlerule[0.8pt]}]
\titleformat{name=\chapter,numberless}[display]
  {\normalfont\centering}
  {}
  {0pt}
  {\huge\bfseries}
  [\vspace{6pt}{\color{rulegray}\titlerule[0.8pt]}]
\titlespacing*{\chapter}{0pt}{10pt}{30pt}

\titleformat{\section}
  {\normalfont\large\bfseries\color{uamobblue}}{\thesection}{0.8em}{}
\titleformat{\subsection}
  {\normalfont\normalsize\bfseries}{\thesubsection}{0.8em}{}

% --- Table columns -------------------------------------------------------
%  Text in narrow table columns is set ragged-right: justified, it opens wide
%  gaps between words and hyphenates short labels. L{width} is the fixed-width
%  version; every tabularx X column gets the same treatment.
\newcolumntype{L}[1]{>{\raggedright\arraybackslash}p{#1}}
%  For cells that open with a list: treat the cell like the top of a minipage,
%  so the first item sits on the same line as the row's label instead of one
%  line below it.
\makeatletter
\newcommand{\cellstartslist}{\@minipagetrue}
\makeatother
\renewcommand{\tabularxcolumn}[1]{>{\raggedright\arraybackslash}p{#1}}

% --- Line breaking -------------------------------------------------------
%  Long identifiers set in typewriter (model names, column names) cannot be
%  hyphenated; a little emergency stretch lets TeX break before them instead of
%  pushing them into the margin.
\setlength{\emergencystretch}{2em}

% --- Running headers -----------------------------------------------------
%  fancyhdr warns if the header box is shorter than its content; 15pt clears
%  the 12pt body font without leaving a gap.
\setlength{\headheight}{15pt}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\nouppercase{\leftmark}}
\fancyhead[R]{\small\thepage}
\renewcommand{\headrulewidth}{0.4pt}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[C]{\small\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
}
